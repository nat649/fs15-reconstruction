cmake_minimum_required(VERSION 3.20)
project(FS15PortableCore LANGUAGES CXX)
add_library(FS15PortableCore STATIC rle8.cpp rle16.cpp tire_forces.cpp wheel_queries.cpp bit_operations.cpp transform.cpp scene_math.cpp scene_graph.cpp scene_state.cpp scene_trs.cpp scene_lifecycle.cpp scene_attributes.cpp shared_scene.cpp animation.cpp animation_resources.cpp animation_xml.cpp scene_properties.cpp scene_xml.cpp scene_owned.cpp scene_lod.cpp)
target_compile_features(FS15PortableCore PUBLIC cxx_std_17)
target_compile_definitions(FS15PortableCore PUBLIC FS15_CORE_STATIC)
target_include_directories(FS15PortableCore PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_compile_options(FS15PortableCore PRIVATE /W4 /WX /utf-8 /fp:strict)
else()
  target_compile_options(FS15PortableCore PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
