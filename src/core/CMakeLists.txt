cmake_minimum_required(VERSION 3.20)
project(FS15PortableCore LANGUAGES CXX)
add_library(FS15PortableCore STATIC rle8.cpp rle16.cpp tire_forces.cpp wheel_queries.cpp bit_operations.cpp transform.cpp scene_math.cpp scene_graph.cpp scene_state.cpp scene_trs.cpp)
target_compile_features(FS15PortableCore PUBLIC cxx_std_17)
target_compile_definitions(FS15PortableCore PUBLIC FS15_CORE_STATIC)
target_include_directories(FS15PortableCore PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_compile_options(FS15PortableCore PRIVATE /W4 /WX /utf-8 /fp:strict)
else()
  target_compile_options(FS15PortableCore PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(FS15PortableCoreCheck ../../tests/native_core.cpp)
  target_link_libraries(FS15PortableCoreCheck PRIVATE FS15PortableCore)
  add_test(NAME portable_native_core COMMAND FS15PortableCoreCheck)
  add_executable(FS15PortableSceneCheck ../../tests/scene_core.cpp)
  target_link_libraries(FS15PortableSceneCheck PRIVATE FS15PortableCore)
  add_test(NAME portable_scene_core COMMAND FS15PortableSceneCheck)
  add_executable(FS15PortableSceneStateCheck ../../tests/scene_state.cpp)
  target_link_libraries(FS15PortableSceneStateCheck PRIVATE FS15PortableCore)
  add_test(NAME portable_scene_state COMMAND FS15PortableSceneStateCheck)
  add_executable(FS15PortableTRSCheck ../../tests/scene_trs.cpp)
  target_link_libraries(FS15PortableTRSCheck PRIVATE FS15PortableCore)
  add_test(NAME portable_scene_trs COMMAND FS15PortableTRSCheck)
  add_executable(FS15PortableHierarchyCheck ../../tests/scene_hierarchy.cpp)
  target_link_libraries(FS15PortableHierarchyCheck PRIVATE FS15PortableCore)
  add_test(NAME portable_scene_hierarchy COMMAND FS15PortableHierarchyCheck)
endif()
