cmake_minimum_required(VERSION 3.20)
project(FS15GE6LinuxBoot LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
  message(FATAL_ERROR "Original GE6 instruction transport currently requires Linux x86_64.")
endif()
option(FS15_GE6_BOOT_STATIC "Build a self-contained diagnostic for local Linux VM testing" OFF)
if(FS15_GE6_BOOT_STATIC)
  set(OPENSSL_USE_STATIC_LIBS TRUE)
endif()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(FS15GE6LinuxBoot main.cpp)
target_compile_features(FS15GE6LinuxBoot PRIVATE cxx_std_17)
target_compile_options(FS15GE6LinuxBoot PRIVATE -Wall -Wextra -Wpedantic -Werror -maccumulate-outgoing-args)
target_link_libraries(FS15GE6LinuxBoot PRIVATE OpenSSL::Crypto ${CMAKE_DL_LIBS} pthread)
if(FS15_GE6_BOOT_STATIC)
  target_link_options(FS15GE6LinuxBoot PRIVATE -static)
endif()
install(TARGETS FS15GE6LinuxBoot RUNTIME DESTINATION bin)
