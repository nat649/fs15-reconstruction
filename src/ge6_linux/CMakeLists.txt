# Pinned, unmodified third-party audio middleware, not recovered GE6 code.
# Keep LIBTYPE/options scoped here so other repository targets are unaffected.
function(ge6_openal)
  include(FetchContent)
  set(LIBTYPE STATIC)
  set(ALSOFT_UTILS OFF CACHE BOOL "" FORCE)
  set(ALSOFT_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(ALSOFT_TESTS OFF CACHE BOOL "" FORCE)
  set(ALSOFT_INSTALL OFF CACHE BOOL "" FORCE)
  set(ALSOFT_UPDATE_BUILD_VERSION OFF CACHE BOOL "" FORCE)
  set(ALSOFT_REQUIRE_ALSA ON CACHE BOOL "" FORCE)
  set(ALSOFT_REQUIRE_PULSEAUDIO ON CACHE BOOL "" FORCE)
  foreach(backend PIPEWIRE OSS SOLARIS SNDIO PORTAUDIO SDL2 SDL3 WAVE)
    set(ALSOFT_BACKEND_${backend} OFF CACHE BOOL "" FORCE)
  endforeach()
  FetchContent_Declare(ge6_openal_source
    URL https://openal-soft.org/openal-releases/openal-soft-1.24.3.tar.bz2
    URL_HASH SHA256=cb5e6197a1c0da0edcf2a81024953cc8fa8545c3b9474e48c852af709d587892)
  FetchContent_MakeAvailable(ge6_openal_source)
  # LGPL static relinking materials accompany every installed boot executable.
  set(relink share/fs15-reconstruction/ge6-audio-relink)
  install(FILES $<TARGET_FILE:OpenAL> DESTINATION ${relink})
  install(DIRECTORY ${ge6_openal_source_SOURCE_DIR}/ DESTINATION ${relink}/openal-soft-1.24.3)
endfunction()
ge6_openal()

set(ge6_x11_default ON)
if(FS15_GE6_BOOT_STATIC)
  set(ge6_x11_default OFF)
endif()
option(FS15_GE6_X11 "Use real X11/RandR display and window transport" ${ge6_x11_default})
set(ge6_link_mode dynamic)
if(FS15_GE6_BOOT_STATIC)
  set(ge6_link_mode static)
endif()
if(FS15_GE6_X11)
  if(FS15_GE6_BOOT_STATIC)
    message(FATAL_ERROR "The X11/GLX transport requires a dynamic Linux build.")
  endif()
  set(ge6_link_mode x11)
  find_package(X11 REQUIRED COMPONENTS Xrandr)
  find_package(OpenGL REQUIRED)
  target_compile_definitions(FS15GE6LinuxBoot PRIVATE GE6_X11=1)
  target_link_libraries(FS15GE6LinuxBoot PRIVATE X11::X11 X11::Xrandr OpenGL::GL)
  add_executable(FS15GE6X11Check ../../tests/check_ge6_x11.cpp)
  target_include_directories(FS15GE6X11Check PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(FS15GE6X11Check PRIVATE GE6_X11=1)
  target_compile_features(FS15GE6X11Check PRIVATE cxx_std_17)
  target_compile_options(FS15GE6X11Check PRIVATE -Wall -Wextra -Wpedantic -Werror -maccumulate-outgoing-args)
  target_link_libraries(FS15GE6X11Check PRIVATE PNG::PNG X11::X11 X11::Xrandr OpenGL::GL pthread)
  add_executable(FS15GE6X11Snapshot x11_snapshot.cpp)
  target_compile_definitions(FS15GE6X11Snapshot PRIVATE GE6_X11=1)
  target_compile_features(FS15GE6X11Snapshot PRIVATE cxx_std_17)
  target_compile_options(FS15GE6X11Snapshot PRIVATE -Wall -Wextra -Wpedantic -Werror -maccumulate-outgoing-args)
  target_link_libraries(FS15GE6X11Snapshot PRIVATE PNG::PNG X11::X11 X11::Xrandr pthread)
endif()
