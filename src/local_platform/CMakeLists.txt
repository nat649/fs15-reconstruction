cmake_minimum_required(VERSION 3.20)
project(FS15LocalPlatform LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The local GE6 platform experiment currently targets Windows x64.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The observed GE6 local-platform ABI requires an x64 build.")
endif()
add_library(fs15_local SHARED local_platform.cpp)
target_compile_features(fs15_local PRIVATE cxx_std_17)
target_compile_definitions(fs15_local PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set_property(TARGET fs15_local PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_options(fs15_local PRIVATE /W4 /WX /utf-8)
