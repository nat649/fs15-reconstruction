find_package(SDL2 REQUIRED)
find_package(OpenGL REQUIRED)
add_executable(FS15GE6Objects main.cpp)
target_compile_features(FS15GE6Objects PRIVATE cxx_std_17)
target_link_libraries(FS15GE6Objects PRIVATE FS15PortableCore SDL2::SDL2 OpenGL::GL)
if(MSVC)
  target_compile_options(FS15GE6Objects PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(FS15GE6Objects PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
install(TARGETS FS15GE6Objects RUNTIME DESTINATION bin)
