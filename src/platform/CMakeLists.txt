add_library(FS15FileTransport STATIC file_transport.cpp)
target_compile_features(FS15FileTransport PUBLIC cxx_std_17)
target_include_directories(FS15FileTransport PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_SOURCE_DIR}/src/core")
target_compile_definitions(FS15FileTransport PUBLIC FS15_CORE_STATIC)
# Keep this small static dependency usable by the standalone Windows menu
# without introducing the complete core DLL or a mismatched MSVC CRT.
target_sources(FS15FileTransport PRIVATE ../core/file_stream.cpp)
target_sources(FS15FileTransport PRIVATE ../core/buffered_file.cpp)
target_sources(FS15FileTransport PRIVATE ../core/memory_file.cpp)
if(MSVC)
  target_compile_options(FS15FileTransport PRIVATE /W4 /WX /utf-8)
  set_property(TARGET FS15FileTransport PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(FS15FileTransport PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(FS15FileTransportCheck ../../tests/file_transport.cpp)
  target_link_libraries(FS15FileTransportCheck PRIVATE FS15FileTransport)
  if(MSVC)
    set_property(TARGET FS15FileTransportCheck PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_test(NAME portable_file_transport COMMAND FS15FileTransportCheck)
endif()
