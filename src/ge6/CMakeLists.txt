include("${CMAKE_SOURCE_DIR}/cmake/ge6-bullet.cmake")

set_target_properties(FS15GE6Physics PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(FS15GE6Physics PUBLIC cxx_std_17)
