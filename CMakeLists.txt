cmake_minimum_required(VERSION 3.20)
project(FS15Reconstruction LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The reconstructed launcher targets Windows.")
endif()
add_library(FS15Core SHARED src/core/rle8.cpp)
target_compile_features(FS15Core PRIVATE cxx_std_17)
add_executable(FS15LauncherRebuilt src/launcher/main.cpp)
target_compile_features(FS15LauncherRebuilt PRIVATE cxx_std_17)
target_compile_definitions(FS15LauncherRebuilt PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(FS15LauncherRebuilt PRIVATE xmllite shlwapi ole32)
option(FS15_BUILD_MENU "Build recovered Lua main-menu runtime when LuaJIT is installed" ON)
if(FS15_BUILD_MENU AND EXISTS "${CMAKE_SOURCE_DIR}/tools/LuaJIT/src/lua51.lib")
  add_executable(FS15MenuRebuilt WIN32 src/menu/main.cpp src/menu/xml_bridge.cpp src/menu/render.cpp src/menu/savegames.cpp)
  target_compile_features(FS15MenuRebuilt PRIVATE cxx_std_17)
  target_compile_definitions(FS15MenuRebuilt PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_include_directories(FS15MenuRebuilt PRIVATE "${CMAKE_SOURCE_DIR}/tools/LuaJIT/src")
  target_link_libraries(FS15MenuRebuilt PRIVATE "${CMAKE_SOURCE_DIR}/tools/LuaJIT/src/lua51.lib" xmllite shlwapi ole32 shell32 uuid gdiplus)
  if(MSVC)
    target_compile_options(FS15MenuRebuilt PRIVATE /W4 /utf-8)
    set_property(TARGET FS15MenuRebuilt PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_executable(FS15MenuMetadataCheck tests/menu_metadata.cpp src/menu/xml_bridge.cpp src/menu/savegames.cpp)
  target_compile_features(FS15MenuMetadataCheck PRIVATE cxx_std_17)
  target_compile_definitions(FS15MenuMetadataCheck PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_include_directories(FS15MenuMetadataCheck PRIVATE src/menu "${CMAKE_SOURCE_DIR}/tools/LuaJIT/src")
  target_link_libraries(FS15MenuMetadataCheck PRIVATE "${CMAKE_SOURCE_DIR}/tools/LuaJIT/src/lua51.lib" xmllite shlwapi ole32)
  if(MSVC)
    target_compile_options(FS15MenuMetadataCheck PRIVATE /W4 /utf-8)
    set_property(TARGET FS15MenuMetadataCheck PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_test(NAME menu_metadata_read_only COMMAND FS15MenuMetadataCheck "${CMAKE_BINARY_DIR}/test-fixtures")
  if(EXISTS "${CMAKE_SOURCE_DIR}/recovered/lua/dataS/scripts/gui/MainScreen.lua")
    add_test(NAME recovered_menu_boot COMMAND FS15MenuRebuilt --root "${CMAKE_SOURCE_DIR}" --smoke --test-inputs)
  endif()
endif()
if(MSVC)
  target_compile_options(FS15LauncherRebuilt PRIVATE /W4 /utf-8)
  set_property(TARGET FS15LauncherRebuilt PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(FS15Core PRIVATE /W4 /utf-8)
  set_property(TARGET FS15Core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
enable_testing()
add_test(NAME launcher_behavior COMMAND FS15LauncherRebuilt --self-test)
find_package(Python3 3.11 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME gar_parser COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests -p test_gar.py)
  set_tests_properties(gar_parser PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  if(EXISTS "${CMAKE_SOURCE_DIR}/tools/LuaJIT/src/luajit.exe" AND EXISTS "${CMAKE_SOURCE_DIR}/tools/luajit-decompiler/main.py")
    add_test(NAME ljd_repeated_literals COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/check_ljd_literals.py")
  endif()
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/inputs/reference/rle8-reference.bin")
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME rle8_native_differential COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_SOURCE_DIR}/scripts/check_rle.py" --dll "$<TARGET_FILE:FS15Core>")
endif()
